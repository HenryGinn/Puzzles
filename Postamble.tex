% \for example the below draws a distance label of 10 that is 0.2 above the line AB.

\tikzset{
	pics/.cd,
	dimension/.style args={#1#2#3#4}{
		code={
			\draw[|<->|] ($(#1)!#3!-90:(#2)$) -- node[midway, fill=white] {#4} ($(#2)!#3!90:(#1)$);
		}
	}
}
